# 2. 既有 index 補上新欄位（把 burp-log-<project_id> 換成你的 index 名稱）
#    新增欄位允許；既有欄位的型別與 index 設定不能改
PUT burp-log-<project_id>/_mapping
{
  "properties": {
    "request":   { "properties": { "hashes_skipped": { "type": "boolean" } } },
    "response":  { "properties": { "hashes_skipped": { "type": "boolean" } } },
    "websocket": { "properties": { "hashes_skipped": { "type": "boolean" },
                                    "body_skip_reason": { "type": "keyword" } } }
  }
}

# 3. 確認（應該看得到 hashes_skipped）
GET burp-log-<project_id>/_mapping/field/request.hashes_skipped
